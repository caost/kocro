**Goal:** 승인된 상단 탭, 접이식 매크로 카드, VIA 형식 토큰 편집기, 최근 실행과 표준 About 패널을 기존 저장·권한·실행 의미를 바꾸지 않고 구현한다.

**Architecture:** `SettingsViewModel`이 펼침, 삭제 취소, 항목별 초안 상태를 소유하고 SwiftUI 뷰는 이 상태를 렌더링한다. `KeyRecorder.swift` 안에 순수 토큰 코덱과 자동완성 모델을 추가해 직접 입력과 AppKit 키 기록이 같은 `ShortcutDefinition`/`TrailingKey` 변환 경로를 사용하게 한다. 메뉴 표시 모델은 실행 UUID를 저장된 런타임 설정의 사용자 제목과 결합하며, About 패널 구성은 테스트 가능한 값 생성과 AppKit 호출을 분리한다.

**Tech Stack:** Swift 5, SwiftUI, AppKit, XCTest, 기존 `Kocro.xcodeproj`(새 외부 의존성 없음)

`MacKeyCodePolicy`의 기존 key code 집합을 그대로 사용해 `tokenEntries(mode:)`를 단일 토큰 조회 API로 추가한다. 이름은 `{KC_A}`, `{KC_1}`, `{KC_ENTER}`, `{KC_SPACE}`, `{KC_TAB}`, `{KC_ESC}`, `{KC_BSPC}`, `{KC_DEL}`, `{KC_LEFT}` 같은 VIA 표기를 사용하고, `{KC_LCMD}`/`{KC_RCMD}` 등 좌우 별칭은 `{KC_CMD}`로 합친다. F1~F20은 기존 `keyCode(forFunction:)`, F21~F24는 실행 단축키 선택 전용 값으로 생성한다.

```swift
enum TokenEditorMode: Equatable { case shortcut, trailing }

struct TokenIssue: Equatable {
    let token: String
    let range: Range<String.Index>?
    let message: String
}

enum TokenCodecError: Error, Equatable {
    case invalid([TokenIssue])
}

enum TokenStoredValue: Equatable {
    case shortcut(ShortcutDefinition)
    case trailing(TrailingKey?)
}

struct TokenValidation: Equatable {
    let source: String
    let canonicalText: String?
    let value: TokenStoredValue?
    let issues: [TokenIssue]
}

enum TokenShortcutCodec {
    static func validate(_ source: String, mode: TokenEditorMode) -> TokenValidation
    static func parse(_ source: String, mode: TokenEditorMode) throws -> TokenValidation
    static func format(_ value: TokenStoredValue) -> String
    static func completions(for fragment: String, mode: TokenEditorMode) -> [String]
}
```

`MacKeyCodePolicy`에는 기존 `characterKeys`, `shortcutEditingKeys`, `editingAndNavigationKeys`, `keypadKeys`, `functionNumbersByKeyCode`를 합성하는 `tokenEntries(mode:)`를 추가한다. 문자·숫자·기호는 `KC_A`, `KC_1`, `KC_MINUS`처럼, 편집·탐색 키는 `KC_ENTER`, `KC_SPACE`, `KC_TAB`, `KC_ESC`, `KC_BSPC`, `KC_DEL`, `KC_LEFT`처럼, keypad는 `KC_KP_0`처럼 canonical token을 부여한다. F1~F20은 `KC_Fn`, 실행 모드에만 F21~F24를 추가한다. `TokenEntry`는 canonical 이름, aliases, `ShortcutKey`를 가지며 `keyNames`와 같은 key code 표에서 생성하므로 지원 범위가 기존 정책과 달라지지 않는다.

코덱 본문은 다음 흐름으로 고정한다. 이 코드는 메서드 선언의 생략 부분을 구현할 때 그대로 사용한다.

```swift
static func validate(_ source: String, mode: TokenEditorMode) -> TokenValidation {
    if source.trimmingCharacters(in: .whitespaces).isEmpty {
        let value: TokenStoredValue = mode == .shortcut
            ? .shortcut(.init(key: .empty, modifiers: [])) : .trailing(nil)
        return .init(source: source, canonicalText: "", value: value, issues: [])
    }
    let pieces = source.split(separator: "+", omittingEmptySubsequences: false)
    var modifiers: ModifierSet = []
    var base: ShortcutKey?
    var baseToken: String?
    var issues: [TokenIssue] = []
    var cursor = source.startIndex

    for piece in pieces {
        let raw = String(piece)
        let trimmed = raw.trimmingCharacters(in: .whitespaces)
        let start = source[cursor...].range(of: raw)?.lowerBound ?? cursor
        let end = source.index(start, offsetBy: raw.count)
        cursor = end < source.endIndex ? source.index(after: end) : end
        let range = start..<end

        if let modifier = modifierAliases[trimmed.uppercased()] {
            modifiers.insert(modifier)
        } else if let entry = MacKeyCodePolicy.tokenEntries(mode: mode)
            .first(where: { $0.matches(trimmed) }) {
            if base != nil {
                issues.append(.init(token: trimmed, range: range,
                    message: "기준 키는 하나만 입력하세요"))
            } else {
                base = entry.key
                baseToken = entry.canonical
            }
        } else {
            issues.append(.init(token: trimmed, range: range,
                message: "지원하는 {KC_...} 토큰으로 수정하세요"))
        }
    }

    if base == nil {
        issues.append(.init(token: source, range: nil, message: "기준 키를 입력하세요"))
    }
    if mode == .shortcut, let base {
        if case .function(let number) = base, (21...24).contains(number), !modifiers.isEmpty {
            issues.append(.init(token: baseToken ?? source, range: nil,
                message: "F21~F24는 보조 키 없이 사용하세요"))
        } else if !isFunction(base, in: 13...24), modifiers.isEmpty {
            issues.append(.init(token: baseToken ?? source, range: nil,
                message: "일반 키에는 보조 키를 하나 이상 추가하세요"))
        }
    }
    if mode == .trailing, let base, isFunction(base, in: 21...24) {
        issues.append(.init(token: baseToken ?? source, range: nil,
            message: "후속 키에는 F21~F24를 사용할 수 없습니다"))
    }
    guard issues.isEmpty, let base, let baseToken else {
        return .init(source: source, canonicalText: nil, value: nil, issues: issues)
    }

    let modifierText = canonicalModifiers(modifiers).joined(separator: "+")
    let canonical = [modifierText, baseToken].filter { !$0.isEmpty }.joined(separator: "+")
    let value: TokenStoredValue
    if mode == .shortcut {
        value = .shortcut(.init(key: base, modifiers: modifiers))
    } else {
        value = .trailing(trailingValue(base: base, modifiers: modifiers))
    }
    return .init(source: source, canonicalText: canonical, value: value, issues: [])
}

static func parse(_ source: String, mode: TokenEditorMode) throws -> TokenValidation {
    let result = validate(source, mode: mode)
    guard result.issues.isEmpty else { throw TokenCodecError.invalid(result.issues) }
    return result
}

static func format(_ value: TokenStoredValue) -> String {
    switch value {
    case .shortcut(let shortcut):
        return canonicalText(key: shortcut.key, modifiers: shortcut.modifiers)
    case .trailing(nil): return ""
    case .trailing(.enter?): return "{KC_ENTER}"
    case .trailing(.space?): return "{KC_SPACE}"
    case .trailing(.tab?): return "{KC_TAB}"
    case .trailing(.custom(let keyCode?, let modifiers)?):
        let key: ShortcutKey = MacKeyCodePolicy.functionNumber(for: keyCode)
            .map(ShortcutKey.function) ?? .keyCode(keyCode)
        return canonicalText(key: key, modifiers: modifiers)
    case .trailing(.customFunction(let number)?):
        return canonicalText(key: .function(number), modifiers: [])
    case .trailing(.custom(keyCode: nil, modifiers: _)?): return ""
    }
}

static func completions(for fragment: String, mode: TokenEditorMode) -> [String] {
    let needle = fragment.uppercased()
    return (canonicalModifierTokens + MacKeyCodePolicy.tokenEntries(mode: mode).map(\.canonical))
        .filter { $0.hasPrefix(needle) }
        .sorted { $0.localizedStandardCompare($1) == .orderedAscending }
}
```

`trailingValue`는 key code 36/49/48의 **보조 키 없는 값만** 각각 `.enter/.space/.tab`으로 만들고, F1~F20은 `MacKeyCodePolicy.keyCode(forFunction:)`으로 변환한 `.custom(keyCode:modifiers:)`, 나머지도 `.custom(keyCode:modifiers:)`로 만든다. 현재 `SettingsValidator`와 `EventBatchFactory`가 거부하는 `.customFunction`을 새 값으로 만들지 않는다. 보조 키가 붙은 Enter/Space/Tab도 preset이 아니라 `.custom`으로 보존한다. `canonicalText`는 `Control, Option, Shift, Command` 순서의 `{KC_CTRL}`, `{KC_OPT}`, `{KC_SHIFT}`, `{KC_CMD}`와 기준 키 하나를 결합한다. `{KC_LCMD}`/`{KC_RCMD}`, `{KC_LCTL}`/`{KC_RCTL}`, `{KC_LALT}`/`{KC_RALT}`, `{KC_LSFT}`/`{KC_RSFT}`는 각 canonical modifier의 alias로 처리한다.

```swift
enum CompletionMove { case up, down }

struct TokenEditorDraft: Equatable {
    private(set) var text: String
    private(set) var issues: [TokenIssue] = []
    private(set) var completions: [String] = []
    private(set) var selectedCompletion = 0
    private(set) var value: TokenStoredValue
    private let baselineText: String
    let mode: TokenEditorMode
    var isDirty: Bool { text != baselineText }

    init(value: TokenStoredValue, mode: TokenEditorMode) {
        self.value = value
        self.mode = mode
        text = TokenShortcutCodec.format(value)
        baselineText = text
    }

    mutating func updateText(_ source: String) {
        text = source
        issues = []
        let fragment = source.split(separator: "+", omittingEmptySubsequences: false).last
            .map(String.init) ?? ""
        completions = fragment.hasPrefix("{")
            ? TokenShortcutCodec.completions(for: fragment, mode: mode) : []
        selectedCompletion = 0
    }

    @discardableResult mutating func commit() -> Bool {
        if mode == .trailing && text.trimmingCharacters(in: .whitespaces).isEmpty {
            value = .trailing(nil); issues = []; completions = []; return true
        }
        let result = TokenShortcutCodec.validate(text, mode: mode)
        issues = result.issues
        guard let parsed = result.value, let canonical = result.canonicalText else { return false }
        value = parsed; text = canonical; completions = []; return true
    }

    mutating func moveCompletion(_ move: CompletionMove) {
        guard !completions.isEmpty else { return }
        let delta = move == .down ? 1 : -1
        selectedCompletion = (selectedCompletion + delta + completions.count) % completions.count
    }

    mutating func acceptCompletion() {
        guard completions.indices.contains(selectedCompletion) else { return }
        let prefix = text.split(separator: "+", omittingEmptySubsequences: false).dropLast()
        text = (prefix.map(String.init) + [completions[selectedCompletion]]).joined(separator: "+")
        completions = []; selectedCompletion = 0
    }

    mutating func selectToken(_ token: String) { updateText(token); _ = commit() }
    mutating func applyRecorded(_ shortcut: ShortcutDefinition) {
        value = mode == .shortcut ? .shortcut(shortcut)
            : .trailing(TokenShortcutCodec.trailingValue(shortcut))
        text = TokenShortcutCodec.format(value); issues = []; completions = []
    }
    mutating func clear() {
        value = mode == .shortcut ? .shortcut(.init(key: .empty, modifiers: [])) : .trailing(nil)
        text = ""; issues = []; completions = []
    }
}
```

`TokenShortcutCodec.trailingValue(_:)`는 기록된 key code 36/49/48의 보조 키 없는 값을 `.enter/.space/.tab`, F1~F20과 나머지 허용 키를 `.custom(keyCode:modifiers:)`로 바꾼다. 이 메서드를 직접 입력의 `trailingValue(base:modifiers:)`와 기록 경로가 함께 사용한다.

enum MacroFieldFocus: Hashable { case title(UUID), shortcut(UUID), text(UUID), trailing(UUID) }
enum MacroStatusBadge: Equatable {
    case inactive, unsaved, registered, conflict, permissionRequired, registrationFailed
    var label: String {
        switch self {
        case .inactive: return "비활성"
        case .unsaved: return "저장 전"
        case .registered: return "등록됨"
        case .conflict: return "충돌"
        case .permissionRequired: return "권한 필요"
        case .registrationFailed: return "등록 실패"
        }
    }
    static func resolve(isEnabled: Bool, isDirty: Bool,
                        registration: RegistrationState?) -> Self {
        if registration == .registrationFailed { return .conflict }
        if registration == .hidStartFailed { return .registrationFailed }
        if registration == .inputMonitoringRequired { return .permissionRequired }
        if isDirty { return .unsaved }
        if !isEnabled { return .inactive }
        return registration == .registered ? .registered : .registrationFailed
    }
}

    deletedMacros.append(.init(macro: settings.macros.remove(at: index), index: index))
    if expandedMacroID == id { expandedMacroID = nil }
}

    for id in offsets.sorted(by: >).compactMap({ settings.macros.indices.contains($0) ? settings.macros[$0].id : nil }) {
        delete(id: id)
    }
}

    guard let deletion = deletedMacros.popLast() else { return }
    settings.macros.insert(deletion.macro, at: min(deletion.index, settings.macros.count))
}

    let tokenDirty = tokenDrafts[.shortcut(id)]?.isDirty == true
        || tokenDrafts[.trailing(id)]?.isDirty == true

    let label: String; let value: String; let help: String; let announcement: String?

        label = fieldLabel; value = draft.text.isEmpty ? "설정 안 됨" : draft.text
        if let issue = draft.issues.first {
            help = issue.message; announcement = issue.message
        } else if draft.completions.indices.contains(draft.selectedCompletion) {
            help = "자동완성 선택 \(draft.completions[draft.selectedCompletion])"
            announcement = help
        } else { help = "토큰을 직접 입력하거나 키로 기록하세요"; announcement = nil }
    }
}

`접힘`에서 `펼쳐짐`으로 바뀌었으며, 같은 카드의 활성화 checkbox를 선택해도 펼침
상태를 유지했다. `{KC_CMD}+{KC_NOPE}` 입력 뒤에는 원문과 함께
`문제 토큰: {KC_NOPE}`가 화면과 접근성 오류에 표시됐다.

첫 실행에서는 카드 12개에 18개를 추가해 접근성 트리에서 30개 행을 확인했다.
드래그 뒤 저장·재실행한 설정 파일의 첫 두 제목은 `매크로 2`, `매크로 1`이었다.
`{KC_F13} + {KC_LCMD}`는 포커스를 옮긴 뒤 `{KC_CMD}+{KC_F13}`으로 바뀌었고,
`{KC_NOTREAL}`은 원문을 유지한 채 오류를 표시했다. 잘못된 토큰으로 저장하면
`표시된 토큰 오류를 수정하세요`가 나타났으며, 수정 후 저장은 성공했다. 등록한
`⌘F13`을 실행한 뒤 메뉴에는 `매크로 1`과 `⌘F13 · 게시 요청 완료 · in 0 seconds`가
표시됐다. About의 GitHub를 선택한 뒤 전면 Chrome 창 제목은
`caost/kocro - Chrome - 재윤 (caost.works)`였다.

- [x] **Step 5: stage 6 checkpoint와 stage 9 formal commit 소유권 확인**

builder는 커밋하지 않는다. conductor는 자동·수동 검증 증거를 확인한 뒤 stage 6 task checkpoint를 만들 수 있다. stage 9 승인 전에는 formal commit을 만들지 않는다. 승인 뒤 conductor는 `wip(task-N):` checkpoint를 정리하고 전체 이슈 변경을 다음 논리 단위로 빠짐없이 구성한다.

1. 토큰 코덱·자동완성·키 기록과 도메인 표시 모델
2. 설정 상단 탭·접이식 카드·초안 편집 수명 주기와 접근성
3. 최근 실행 메뉴·표준 About 패널과 통합 검증

checkpoint에 기록된 파일은 formal commit 재구성 과정에서 누락하지 않는다.
